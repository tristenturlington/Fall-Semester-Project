\section{Background and Prior Literature}
To ground our differential equation model in empirical ecology, we adopt the common milkweed (\textit{Asclepias syriaca}), monarch butterfly caterpillar (\textit{Danaus plexippus}), and bumblebee (\textit{Bombus} spp.) as our tri-trophic model system. Previous theoretical work by Revilla (2015) modeled plant--herbivore interactions under logistic plant growth and linear herbivory; our Stage 2 model extends this framework by incorporating saturating Type II pollinator visitation and density-dependent pollinator competition.

\section{Dimensional Consistency Verification}
We verify that all terms in the Stage 2 system maintain dimensional consistency in units of $\text{plants}\cdot\text{day}^{-1}$, $\text{caterpillars}\cdot\text{day}^{-1}$, and $\text{bees}\cdot\text{day}^{-1}$:

\begin{align}
\frac{dP}{dt} &= \underbrace{rP\left(1-\frac{P}{K}\right)}_{\text{day}^{-1}\cdot\text{plants}} 
                 - \underbrace{aPC}_{(\text{cat}\cdot\text{day})^{-1}\cdot\text{plants}\cdot\text{cat}} 
                 + \underbrace{\beta \frac{vPB}{H+P}\left(1-\frac{P}{K}\right)}_{\frac{\text{plants}}{\text{visit}}\cdot\frac{\text{visits}}{\text{bee}\cdot\text{day}}\cdot\frac{\text{plants}\cdot\text{bees}}{\text{plants}}} 
                 \quad \left[\frac{\text{plants}}{\text{day}}\right] \\[6pt]
\frac{dC}{dt} &= \underbrace{\epsilon a P C}_{\frac{\text{cat}}{\text{plant}}\cdot(\text{cat}\cdot\text{day})^{-1}\cdot\text{plants}\cdot\text{cat}} 
                 - \underbrace{\mu C}_{\text{day}^{-1}\cdot\text{cat}} 
                 \quad \left[\frac{\text{caterpillars}}{\text{day}}\right] \\[6pt]
\frac{dB}{dt} &= \underbrace{\gamma \frac{vPB}{H+P}}_{\frac{\text{bees}}{\text{visit}}\cdot\frac{\text{visits}}{\text{bee}\cdot\text{day}}\cdot\frac{\text{plants}\cdot\text{bees}}{\text{plants}}} 
                 - \underbrace{dB}_{\text{day}^{-1}\cdot\text{bees}} 
                 - \underbrace{qB^2}_{(\text{bee}\cdot\text{day})^{-1}\cdot\text{bees}^2} 
                 \quad \left[\frac{\text{bees}}{\text{day}}\right]
\end{align}

$\epsilon$ & $0.40$ & $\text{cat}\cdot\text{plant}^{-1}$ & \textbf{Caterpillar conversion efficiency:} Accounts for average larval survival ($\sim 58\%$ across nine \textit{Asclepias} species; Pocius et al., 2018) and biomass conversion ($<1$). \\
$\mu$ & $0.069$ & $\text{day}^{-1}$ & \textbf{Caterpillar exit/mortality rate:} Baseline larval stage duration of $\sim 14.5\text{ days}$ ($\mu \approx 1/14.5$; Pocius et al., 2018). \\
$v$ & $2428$ & $\text{visits}\cdot(\text{bee}\cdot\text{day})^{-1}$ & \textbf{Maximum bee visitation rate:} Based on $346.9\ \text{min}\cdot\text{day}^{-1}$ foraging time ($5.9\text{ trips/day} \times 58.5\text{ min/trip}$) at $\sim 7\text{ visits}\cdot\text{min}^{-1}$ (Minahan \& Brunet, 2018). \\

$\beta$ & $0.0005$ & $\text{plants}\cdot\text{visit}^{-1}$ & \textbf{Plant growth per bee visit:} Derived from $0.0149\text{ pods/flower}$ ($0.93\text{ pods} / 62.5\text{ flowers}$; Woods et al., 2012) multiplied by a $1\text{--}5\%$ seedling establishment rate ($0.00015\text{--}0.00075$). \\
$\gamma$ & $2.96 \times 10^{-5}$ & $\text{bees}\cdot\text{visit}^{-1}$ & \textbf{Bee growth per flower visit:} Calibrated to balance natural mortality at carrying capacity ($\gamma = \frac{d}{v \cdot K/(H+K)}$), representing colony resource conversion into new workers. \\
$d$ & $0.048$ & $\text{day}^{-1}$ & \textbf{Natural bee mortality rate:} Inverse of mean bumblebee worker lifespan of $20.95\text{ days}$ (range: $17.6\text{--}30\text{ days}$; Da Silva-Matos \& Gar\'ofalo, 2000). \\
$q$ & $0.0041$ & $(\text{bee}\cdot\text{day})^{-1}$ & \textbf{Density-dependent bee competition:} Calibrated from $q = \left(\gamma \frac{vP}{H+P} - d\right)/B$ for a foraging density of $B=10\text{ bees}$ in a high-density patch ($P=40$). \\

\section{Connection to Equilibrium Thresholds}
Using Bella's analytical coexistence condition,
\begin{equation}
\frac{dH}{\gamma v - d} < \frac{\mu}{\epsilon a} < K,
\end{equation}
our baseline parameters yield an equilibrium plant threshold of $P^* = \frac{\mu}{\epsilon a} = \frac{0.069}{(0.40)(0.00138)} = 125\text{ plants}$. This demonstrates that in a small patch ($K = 50 < 125$), monarch caterpillars cannot sustain a positive equilibrium ($E_{PB}$ is stable instead), whereas in a larger meadow ($K > 125$), three-species coexistence ($E_{PCB}$) becomes biologically feasible.
